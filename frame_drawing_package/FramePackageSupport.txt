Attribute VB_Name = "FramePackageSupport"
Option Explicit

Public Function HasItems(ByVal values As Variant) As Boolean
    On Error GoTo EmptyArray
    If IsArray(values) Then HasItems = (UBound(values) >= LBound(values))
EmptyArray:
End Function

Public Sub Require(ByVal condition As Boolean, ByVal message As String)
    If condition = False Then Err.Raise vbObjectError + 3400, "FrameDrawingPackage", message
End Sub

Public Function SameModel(ByVal host As SldWorks.SldWorks, ByVal first As SldWorks.ModelDoc2, _
                          ByVal second As SldWorks.ModelDoc2) As Boolean
    If first Is Nothing Then Exit Function
    If second Is Nothing Then Exit Function
    If first Is second Then SameModel = True: Exit Function
    SameModel = (host.IsSame(first, second) = swObjectSame)
End Function

Public Sub ActivateDocument(ByVal host As SldWorks.SldWorks, ByVal model As SldWorks.ModelDoc2)
    Dim active As SldWorks.ModelDoc2, errors As Long
    Call Require((model Is Nothing) = False, "Нет документа для активации.")
    Set active = host.ActiveDoc
    If SameModel(host, active, model) Then Exit Sub
    Set active = host.ActivateDoc3(model.GetTitle, False, swDontRebuildActiveDoc, errors)
    Set active = host.ActiveDoc
    Call Require(SameModel(host, active, model), "Не активирован документ: " & model.GetTitle & "; код=" & errors)
End Sub

Public Sub ActivateSheet(ByVal host As SldWorks.SldWorks, ByVal model As SldWorks.ModelDoc2, ByVal sheetName As String)
    Dim drawing As SldWorks.DrawingDoc, sheet As SldWorks.Sheet
    ActivateDocument host, model
    Set drawing = model
    Call Require(drawing.ActivateSheet(sheetName), "Не активирован лист: " & sheetName)
    Set sheet = drawing.GetCurrentSheet
    Call Require((sheet Is Nothing) = False, "Нет текущего листа.")
    Call Require(StrComp(sheet.GetName, sheetName, vbBinaryCompare) = 0, "Не подтверждён текущий лист.")
    drawing.EditSheet
    model.ClearSelection2 True
End Sub

Public Function DrawingComponentFor(ByVal view As SldWorks.View, ByVal requested As SldWorks.Component2) As SldWorks.Component2
    Dim values As Variant, item As SldWorks.DrawingComponent, candidate As SldWorks.Component2
    Dim root As SldWorks.DrawingComponent, rootComponent As SldWorks.Component2
    Dim rootName As String, rootAlias As String, i As Long, matches As Long, sameName As Boolean
    Set root = view.RootDrawingComponent
    If Not root Is Nothing Then
        rootName = root.Name
        Set rootComponent = root.Component
        If Not rootComponent Is Nothing Then rootAlias = rootComponent.Name2
    End If
    values = view.GetVisibleDrawingComponents
    Call Require(HasItems(values), "Вид не вернул видимые компоненты. Проверьте конфигурацию и видимость рамы.")
    For i = LBound(values) To UBound(values)
        Set item = values(i)
        Set candidate = item.Component
        If Not candidate Is Nothing Then
            sameName = (StrComp(candidate.Name2, requested.Name2, vbTextCompare) = 0)
            If Len(rootName) > 0 Then sameName = sameName Or _
                (StrComp(candidate.Name2, rootName & "/" & requested.Name2, vbTextCompare) = 0)
            If Len(rootAlias) > 0 Then sameName = sameName Or _
                (StrComp(candidate.Name2, rootAlias & "/" & requested.Name2, vbTextCompare) = 0)
            If sameName Then
                If StrComp(candidate.GetPathName, requested.GetPathName, vbTextCompare) = 0 Then
                    If StrComp(candidate.ReferencedConfiguration, requested.ReferencedConfiguration, vbTextCompare) = 0 Then
                        matches = matches + 1
                        Set DrawingComponentFor = candidate
                    End If
                End If
            End If
        End If
    Next i
    Call Require(matches = 1, "Экземпляр рамы не найден однозначно в виде: " & requested.Name2 & "; совпадений=" & matches)
End Function

Public Function StandardView(ByVal model As SldWorks.ModelDoc2, ByVal modelPath As String, _
                             ByVal config As String, ByVal direction As String, ByVal viewName As String) As SldWorks.View
    Dim drawing As SldWorks.DrawingDoc, view As SldWorks.View, aliases As Variant, alias As Variant
    Set drawing = model
    Select Case direction
        Case "ISO": aliases = Array("*Isometric", "*Изометрия")
        Case "FRONT": aliases = Array("*Front", "*Спереди")
        Case "TOP": aliases = Array("*Top", "*Сверху")
        Case Else: Call Require(False, "Неизвестная ориентация: " & direction)
    End Select
    For Each alias In aliases
        Set view = drawing.CreateDrawViewFromModelView3(modelPath, CStr(alias), 0.1, 0.1, 0#)
        If Not view Is Nothing Then Exit For
    Next alias
    Call Require((view Is Nothing) = False, "Не создан вид " & direction & ": " & modelPath)
    view.ReferencedConfiguration = config
    Call Require(StrComp(view.ReferencedConfiguration, config, vbTextCompare) = 0, "Не применена конфигурация вида: " & config)
    Call Require(view.SetName2(viewName), "Не присвоено имя вида: " & viewName)
    view.UseSheetScale = False
    model.EditRebuild3
    Call Require((view.ReferencedDocument Is Nothing) = False, "Вид не загрузил связанную модель.")
    Set StandardView = view
End Function

Public Sub FitView(ByVal model As SldWorks.ModelDoc2, ByVal view As SldWorks.View, _
                   ByVal left As Double, ByVal bottom As Double, ByVal right As Double, ByVal top As Double)
    Dim box As Variant, position As Variant, width As Double, height As Double, factor As Double, factorY As Double
    Call Require(right > left And top > bottom, "Нет области для размещения вида.")
    box = ViewBox(view)
    width = box(2) - box(0): height = box(3) - box(1)
    Call Require(width > 0# And height > 0#, "Вырожденный габарит вида.")
    factor = (right - left) / width: factorY = (top - bottom) / height
    If factorY < factor Then factor = factorY
    view.UseSheetScale = False
    view.ScaleDecimal = view.ScaleDecimal * factor
    model.EditRebuild3
    box = ViewBox(view): position = view.Position
    Call Require(HasItems(position), "Не прочитано положение вида.")
    position(0) = position(0) + (left + right - box(0) - box(2)) / 2#
    position(1) = position(1) + (bottom + top - box(1) - box(3)) / 2#
    view.Position = position
    model.EditRebuild3
    box = ViewBox(view)
    Call Require(box(0) >= left - 0.001 And box(2) <= right + 0.001 And _
            box(1) >= bottom - 0.001 And box(3) <= top + 0.001, "Вид не поместился в выделенную область.")
End Sub

Public Sub FitPair(ByVal model As SldWorks.ModelDoc2, ByVal front As SldWorks.View, ByVal top As SldWorks.View, _
                   ByVal width As Double, ByVal height As Double)
    Dim a As Variant, b As Variant, k As Double, current As Double, candidates As Variant, value As Variant
    Dim half As Double, usableWidth As Double, usableHeight As Double
    half = width / 2#: usableWidth = half - 0.065: usableHeight = height - 0.09
    Call Require(usableWidth > 0.04 And usableHeight > 0.06, "Формат листа слишком мал для двух видов и размеров.")
    front.UseSheetScale = False: top.UseSheetScale = False
    front.ScaleDecimal = 1#: top.ScaleDecimal = 1#
    model.EditRebuild3
    a = ViewBox(front): b = ViewBox(top)
    Call Require(a(2) > a(0) And a(3) > a(1) And b(2) > b(0) And b(3) > b(1), "Нет габаритов видов.")
    k = usableWidth / (a(2) - a(0))
    candidates = Array(usableHeight / (a(3) - a(1)), usableWidth / (b(2) - b(0)), usableHeight / (b(3) - b(1)))
    For Each value In candidates
        If CDbl(value) < k Then k = CDbl(value)
    Next value
    front.ScaleDecimal = k: top.ScaleDecimal = k
    model.EditRebuild3
    PlaceLowerLeft model, front, 0.025, 0.05
    PlaceLowerLeft model, top, half + 0.045, 0.05
    Call Require(Abs(front.ScaleDecimal - top.ScaleDecimal) < 0.000000001, "Различаются масштабы переднего и верхнего видов.")
End Sub

Private Sub PlaceLowerLeft(ByVal model As SldWorks.ModelDoc2, ByVal view As SldWorks.View, ByVal x As Double, ByVal y As Double)
    Dim box As Variant, position As Variant
    box = ViewBox(view): position = view.Position
    Call Require(HasItems(position), "Нет положения вида.")
    position(0) = position(0) + x - box(0): position(1) = position(1) + y - box(1)
    view.Position = position
    model.EditRebuild3
    box = ViewBox(view)
    Call Require(Abs(box(0) - x) < 0.001 And Abs(box(1) - y) < 0.001, "Не подтверждено размещение вида.")
End Sub

Public Function ViewBox(ByVal view As SldWorks.View) As Variant
    Dim value As Variant
    value = view.GetOutline
    Call Require(HasItems(value), "Вид не вернул габарит.")
    Call Require(LBound(value) = 0 And UBound(value) >= 3, "Неполный габарит вида.")
    ViewBox = value
End Function

Public Sub AuditCurrentSheet(ByVal model As SldWorks.ModelDoc2, ByVal audit As Collection)
    Dim drawing As SldWorks.DrawingDoc, sheetView As SldWorks.View, view As SldWorks.View
    Dim display As SldWorks.DisplayDimension, dimension As SldWorks.Dimension, record As Variant
    Set drawing = model
    Set sheetView = drawing.GetFirstView
    Call Require((sheetView Is Nothing) = False, "Не прочитан листовой вид.")
    Set view = sheetView.GetNextView
    Do While Not view Is Nothing
        Set display = view.GetFirstDisplayDimension5
        Do While Not display Is Nothing
            Set dimension = display.GetDimension2(0)
            Call Require((dimension Is Nothing) = False, "Не прочитано значение созданного размера.")
            ReDim record(0 To 1)
            Set record(0) = display
            record(1) = dimension.SystemValue
            audit.Add record
            Set display = display.GetNext5
        Loop
        Set view = view.GetNextView
    Loop
End Sub

Public Sub ValidateAudit(ByVal audit As Collection)
    Dim record As Variant, display As SldWorks.DisplayDimension, dimension As SldWorks.Dimension
    Dim annotation As SldWorks.Annotation
    For Each record In audit
        Set display = record(0)
        Set dimension = display.GetDimension2(0)
        Call Require((dimension Is Nothing) = False, "После перестроения не прочитан размер.")
        Call Require(Abs(dimension.SystemValue - CDbl(record(1))) <= 0.000001, "После восстановления конфигураций изменилось значение размера.")
        Set annotation = display.GetAnnotation
        Call Require((annotation Is Nothing) = False, "Нет аннотации размера.")
        Call Require(annotation.IsDangling = False, "После восстановления конфигураций размер потерял привязку.")
    Next record
End Sub

Public Sub ApplyViewDisplayState(ByVal model As SldWorks.ModelDoc2, ByVal view As SldWorks.View, ByVal stateName As String)
    view.DisplayState = stateName
    model.EditRebuild3
    Call Require(StrComp(view.DisplayState, stateName, vbTextCompare) = 0, "Не закреплено состояние вида: " & stateName)
End Sub

Public Sub HideConstructionGeometry(ByVal model As SldWorks.ModelDoc2)
    ' Document-only display settings; do not modify the source assembly or application preferences.
    Call Require(model.SetUserPreferenceToggle(swDisplayAxes, False), "Не скрыты вспомогательные оси.")
    Call Require(model.SetUserPreferenceToggle(swDisplayTemporaryAxes, False), "Не скрыты временные оси.")
    Call Require(model.SetUserPreferenceToggle(swDisplayPlanes, False), "Не скрыты вспомогательные плоскости.")
    Call Require(model.SetUserPreferenceToggle(swDisplayOrigins, False), "Не скрыты начала координат.")
    Call Require(model.SetUserPreferenceToggle(swDisplaySketches, False), "Не скрыты вспомогательные эскизы.")
End Sub

Public Sub SetSheetScale(ByVal model As SldWorks.ModelDoc2, ByVal numerator As Double, ByVal denominator As Double)
    Dim drawing As SldWorks.DrawingDoc, sheet As SldWorks.Sheet, props As Variant
    Set drawing = model: Set sheet = drawing.GetCurrentSheet
    Call Require(sheet.SetScale(numerator, denominator, False, False), "Не применён масштаб листа.")
    props = sheet.GetProperties
    Call Require(Abs(CDbl(props(2)) / CDbl(props(3)) - numerator / denominator) < 0.00000001, "Не подтверждён масштаб листа.")
End Sub

Public Sub SetScaleAndPlace(ByVal model As SldWorks.ModelDoc2, ByVal view As SldWorks.View, _
                            ByVal layoutScale As Double, ByVal x As Double, ByVal y As Double)
    view.UseSheetScale = False: view.ScaleDecimal = layoutScale
    model.EditRebuild3
    PlaceLowerLeft model, view, x, y
    Call Require(Abs(view.ScaleDecimal - layoutScale) < 0.00000001, "Не подтверждён масштаб вида.")
End Sub

Public Sub FormatMountingDimension(ByVal display As SldWorks.DisplayDimension)
    Dim annotation As SldWorks.Annotation, format As SldWorks.TextFormat
    Set annotation = display.GetAnnotation
    Set format = annotation.GetTextFormat(0)
    Call Require((format Is Nothing) = False, "Нет формата текста монтажного размера.")
    format.CharHeight = 0.0035
    Call Require(annotation.SetTextFormat(0, False, format), "Не установлен текст размеров 3,5 мм.")
End Sub

Private Function Smaller(ByVal a As Double, ByVal b As Double) As Double
    Smaller = a
    If b < a Then Smaller = b
End Function

Private Function Padding(ByVal geometry As Variant, ByVal envelope As Variant) As Variant
    Dim left As Double, bottom As Double, right As Double, top As Double
    left = geometry(0) - envelope(0): bottom = geometry(1) - envelope(1)
    right = envelope(2) - geometry(2): top = envelope(3) - geometry(3)
    If left < 0# Then left = 0#
    If bottom < 0# Then bottom = 0#
    If right < 0# Then right = 0#
    If top < 0# Then top = 0#
    Padding = Array(left + right, bottom + top)
End Function

Private Sub MoveEnvelope(ByVal model As SldWorks.ModelDoc2, ByVal view As SldWorks.View, _
                         ByVal envelope As Variant, ByVal x As Double, ByVal y As Double)
    Dim position As Variant
    position = view.Position
    position(0) = position(0) + x - envelope(0): position(1) = position(1) + y - envelope(1)
    view.Position = position
    model.EditRebuild3
End Sub

Public Sub FitMountingPair(ByVal host As SldWorks.SldWorks, ByVal model As SldWorks.ModelDoc2, _
                           ByVal front As SldWorks.View, ByVal top As SldWorks.View, _
                           ByVal frame As SldWorks.Component2, ByVal width As Double, ByVal height As Double)
    Dim a As Variant, b As Variant, ea As Variant, eb As Variant, pa As Variant, pb As Variant
    Dim fw As Double, fh As Double, tw As Double, th As Double, availableW As Double, availableH As Double
    Dim horizontal As Double, vertical As Double, layoutScale As Double, current As Double, gap As Double
    Dim ew1 As Double, eh1 As Double, ew2 As Double, eh2 As Double, pass As Long, sideBySide As Boolean
    availableW = width - 0.035: availableH = height - 0.06: gap = 0.012
    Call Require(availableW > 0# And availableH > 0#, "Недостаточный формат для монтажных видов.")
    SetScaleAndPlace model, front, 0.1, 0.10, 0.10
    SetScaleAndPlace model, top, 0.1, 0.10, 0.10
    For pass = 1 To 3
        current = front.ScaleDecimal
        a = ViewBox(front): b = ViewBox(top)
        ea = FrameAutoSize.MeasureForPackage(host, model, front, frame, "FRONT")
        eb = FrameAutoSize.MeasureForPackage(host, model, top, frame, "TOP")
        pa = Padding(a, ea): pb = Padding(b, eb)
        fw = (a(2) - a(0)) / current: fh = (a(3) - a(1)) / current
        tw = (b(2) - b(0)) / current: th = (b(3) - b(1)) / current
        Call Require(fw > 0# And fh > 0# And tw > 0# And th > 0#, "Пустой габарит монтажных видов.")
        horizontal = Smaller((availableW - gap - pa(0) - pb(0)) / (fw + tw), _
                     Smaller((availableH - pa(1)) / fh, (availableH - pb(1)) / th))
        vertical = Smaller((availableH - gap - pa(1) - pb(1)) / (fh + th), _
                   Smaller((availableW - pa(0)) / fw, (availableW - pb(0)) / tw))
        sideBySide = (horizontal >= vertical)
        If sideBySide Then layoutScale = horizontal Else layoutScale = vertical
        layoutScale = layoutScale * 0.985
        Call Require(layoutScale > 0#, "Для модели и размерных линий недостаточно места внутри рамки.")
        front.ScaleDecimal = layoutScale: top.ScaleDecimal = layoutScale
        model.EditRebuild3
        ea = FrameAutoSize.MeasureForPackage(host, model, front, frame, "FRONT")
        eb = FrameAutoSize.MeasureForPackage(host, model, top, frame, "TOP")
        ew1 = ea(2) - ea(0): eh1 = ea(3) - ea(1): ew2 = eb(2) - eb(0): eh2 = eb(3) - eb(1)
        If sideBySide Then
            MoveEnvelope model, front, ea, 0.025 + (availableW - ew1 - ew2 - gap) / 2#, 0.045 + (availableH - eh1) / 2#
            MoveEnvelope model, top, eb, 0.025 + (availableW - ew1 - ew2 - gap) / 2# + ew1 + gap, 0.045 + (availableH - eh2) / 2#
        Else
            MoveEnvelope model, top, eb, 0.025 + (availableW - ew2) / 2#, 0.045 + (availableH - eh1 - eh2 - gap) / 2#
            MoveEnvelope model, front, ea, 0.025 + (availableW - ew1) / 2#, 0.045 + (availableH - eh1 - eh2 - gap) / 2# + eh2 + gap
        End If
    Next pass
    ea = FrameAutoSize.MeasureForPackage(host, model, front, frame, "FRONT")
    eb = FrameAutoSize.MeasureForPackage(host, model, top, frame, "TOP")
    ValidatePairBoxes ea, eb, width, height
    Debug.Print "Mounting layoutScale=" & layoutScale & "; side-by-side=" & sideBySide
End Sub

Private Sub ValidatePairBoxes(ByVal a As Variant, ByVal b As Variant, ByVal width As Double, ByVal height As Double)
    Call Require(a(0) >= 0.025 And b(0) >= 0.025 And a(1) >= 0.045 And b(1) >= 0.045 And _
        a(2) <= width - 0.01 And b(2) <= width - 0.01 And a(3) <= height - 0.015 And b(3) <= height - 0.015, _
        "Монтажные виды или размеры выходят за рабочую область рамки.")
    Call Require(a(2) + 0.005 <= b(0) Or b(2) + 0.005 <= a(0) Or _
        a(3) + 0.005 <= b(1) Or b(3) + 0.005 <= a(1), "Пересекаются области монтажных видов и размеров.")
End Sub

Private Function MountedEnvelope(ByVal view As SldWorks.View) As Variant
    Dim box As Variant, display As SldWorks.DisplayDimension, annotation As SldWorks.Annotation, p As Variant
    box = ViewBox(view)
    Set display = view.GetFirstDisplayDimension5
    Do While Not display Is Nothing
        Set annotation = display.GetAnnotation
        p = annotation.GetPosition
        Call Require(HasItems(p), "Нет позиции монтажного размера.")
        ' Conservative text allowances for 3.5 mm text, including rotated captions and arrows.
        If p(0) - 0.016 < box(0) Then box(0) = p(0) - 0.016
        If p(0) + 0.016 > box(2) Then box(2) = p(0) + 0.016
        If p(1) - 0.009 < box(1) Then box(1) = p(1) - 0.009
        If p(1) + 0.009 > box(3) Then box(3) = p(1) + 0.009
        Set display = display.GetNext5
    Loop
    MountedEnvelope = box
End Function

Public Sub CheckMountingPair(ByVal model As SldWorks.ModelDoc2, ByVal front As SldWorks.View, ByVal top As SldWorks.View, _
                             ByVal width As Double, ByVal height As Double)
    model.EditRebuild3
    ValidatePairBoxes MountedEnvelope(front), MountedEnvelope(top), width, height
End Sub
